# --description--

Now, show timestamps in the email listing too. Update the `__str__` method to add ` | Time: ` followed by the timestamp, formatted as `'%Y-%m-%d %H:%M'`, after the subject:

```md
[status] From: sender | Subject: subject | Time: time
```

To keep the `Time` column lined up, left-align the subject in a width of `15`. A width is only a minimum, so a longer subject would still print in full and push `Time` out of line. Slice the subject to its first `15` characters before formatting it to prevent that.

For example, an email with a short subject and one with a long subject would be:

```md
[Unread] From: Alice      | Subject: Hello           | Time: 2025-01-15 09:30
[Unread] From: Alice      | Subject: Quarterly budge | Time: 2025-01-15 09:30
```

import datetime

class Email:
    def __init__(self, sender, receiver, subject, body):
        self.sender = sender
        self.receiver = receiver
        self.subject = subject
        self.body = body
        self.timestamp = datetime.datetime.now()
        self.read = False

    def mark_as_read(self):
        self.read = True

    def display_full_email(self):
        self.mark_as_read()
        print(f"\n{' Email ':-^29}")
        print(f"{'From:':<10}{self.sender.name}")
        print(f"{'To:':<10}{self.receiver.name}")
        print(f"{'Subject:':<10}{self.subject}")
        print(f"{'Received:':<10}{self.timestamp.strftime('%Y-%m-%d %H:%M')}")
        print(f"{'Body:':<10}{self.body}")
        print(f"{'':-^29}\n")

    def __str__(self):
        status = 'Read' if self.read else 'Unread'
--fcc-editable-region--
        return f'[{status:<6}] From: {self.sender.name:<10} | Subject: {self.subject}'
--fcc-editable-region--

class User:
    def __init__(self, name):
        self.name = name
        self.inbox = Inbox()

    def send_email(self, receiver, subject, body):
        email = Email(sender=self, receiver=receiver, subject=subject, body=body)
        receiver.inbox.receive_email(email)

class Inbox:
    def __init__(self):
        self.emails = []

    def receive_email(self, email):
        self.emails.append(email)

    def list_emails(self):
        if not self.emails:
            print('Your inbox is empty.\n')
            return
        print('\nYour Emails:')
        for i, email in enumerate(self.emails, start=1):
            print(f'{i:>2}. {email}')

    def read_email(self, index):
        if not self.emails:
            print('Inbox is empty.\n')
            return
        actual_index = index - 1
        if actual_index < 0 or actual_index >= len(self.emails):
            print('Invalid email number.\n')
            return
        self.emails[actual_index].display_full_email()

    def delete_email(self, index):
        if not self.emails:
            print('Inbox is empty.\n')
            return
        actual_index = index - 1
        if actual_index < 0 or actual_index >= len(self.emails):
            print('Invalid email number.\n')
            return
        del self.emails[actual_index]
        print('Email deleted.\n')
