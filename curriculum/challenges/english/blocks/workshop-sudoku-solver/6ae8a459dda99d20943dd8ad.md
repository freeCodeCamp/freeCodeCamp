# --description--

The third part is to undo, and it's the key step of backtracking. If the recursive call returns `False`, the number you placed led to a dead end somewhere further down, so it can't be part of the solution.

After your `if solve(board)` statement, reset the cell by assigning `0` to `board[row][col]`. This way, the loop can try the next number in a clean cell. Without this line, the wrong number would stay on the board.

def print_board(board):
    for row in board:

def find_empty_cell(board):
    for row in range(9):
        for col in range(9):
            if board[row][col] == 0:
                return row, col
    return None


def is_valid(board, row, col, num):
    if num in board[row]:
        return False

    for r in range(9):
        if board[r][col] == num:
            return False

    box_row = (row // 3) * 3
    box_col = (col // 3) * 3
    for r in range(box_row, box_row + 3):
        for c in range(box_col, box_col + 3):
            if board[r][c] == num:
                return False

    return True


def solve(board):
    empty_cell = find_empty_cell(board)
    if empty_cell is None:
        return True

    row, col = empty_cell

    for num in range(1, 10):
        if is_valid(board, row, col, num):
            board[row][col] = num
            if solve(board):
                return True
